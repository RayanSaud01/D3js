HTMLBar = 
-- 1. Context & Revenue
VAR SelectedYear = SELECTEDVALUE('Sales'[Order Date].[Year])
VAR CurrentRegion = SELECTEDVALUE('Sales'[Region])
VAR Revenue = SUM('Sales'[Total Revenue])
VAR YearIsFiltered = ISFILTERED('Sales'[Order Date].[Year])

-- 2. Prior Year Calculation
VAR RevenuePY = 
    IF(NOT(YearIsFiltered), 
        BLANK(),
        CALCULATE(
            SUM('Sales'[Total Revenue]),
            FILTER(
                ALL('Sales'),
                'Sales'[Region] = CurrentRegion &&
                'Sales'[Order Date].[Year] = SelectedYear - 1
            )
        )
    )

-- 3. GLOBAL SCALE
VAR AllRegionsInView = ALLSELECTED('Sales'[Region])
VAR MaxCurrentVisible = MAXX(AllRegionsInView, CALCULATE(SUM('Sales'[Total Revenue])))
VAR MaxValueForScale = 
    IF(YearIsFiltered,
        MAX(MaxCurrentVisible, MAXX(AllRegionsInView, CALCULATE(SUM('Sales'[Total Revenue]), FILTER(ALL('Sales'), 'Sales'[Order Date].[Year] = SelectedYear - 1)))),
        MaxCurrentVisible
    )
VAR GlobalMax = MaxValueForScale * 1.1

-- 4. LOCAL FORMATTING
VAR Val = ABS(Revenue)
VAR Sfx = IF(Val >= 1e9, "B", IF(Val >= 1e6, "M", IF(Val >= 1e3, "K", "")))
VAR Dvsr = IF(Val >= 1e9, 1e9, IF(Val >= 1e6, 1e6, IF(Val >= 1e3, 1e3, 1)))

-- 5. SVG Internal Coordinates
VAR V_Width = 400  
VAR V_Height = 65  
VAR BarMaxVWidth = 390 
VAR BarWidth = DIVIDE(Revenue, GlobalMax) * BarMaxVWidth
VAR PYMarkerPos = DIVIDE(RevenuePY, GlobalMax) * BarMaxVWidth

-- 6. Variance & Styling
VAR VarPct = DIVIDE(Revenue - RevenuePY, RevenuePY)
VAR VarColor = IF(VarPct >= 0, "#228B22", "#D22B2B")
VAR BarColor = IF(YearIsFiltered && VarPct < 0, "#E07A5F", "#457B9D")

-- 7. HTML COMPONENTS (Hierarchy & Consistency)
VAR HeaderHTML = 
    "<div style='font-family:Segoe UI, Arial; padding: 20px 15px; background-color:#FDFBF7; border-bottom:2px solid #EEE;'> " &
        "<div style='font-size:22px; font-weight:900; color:#2C3E50; letter-spacing:-0.5px;'>Sales Performance</div>" &
        "<div style='font-size:12px; font-weight:600; color:#7F8C8D; text-transform: uppercase; letter-spacing:1px; margin-top:4px;'>" & 
            IF(YearIsFiltered, "Fiscal Year " & SelectedYear & " • Comparison to PY", "Historical Aggregate Analysis") & 
        "</div>" &
    "</div>"

VAR ResponsiveRow = 
    "<div style='display: flex; align-items: center; background-color:#FDFBF7; padding: 12px 15px; border-bottom: 1px solid #F0EAE0;'>" &
        
        -- COLUMN 1: Visual Chart Area
        "<div style='flex: 1;'>" &
            "<svg viewBox='0 0 " & V_Width & " " & V_Height & "' width='100%' preserveAspectRatio='xMinYMid meet' style='display:block;'>" &
                -- REGION LABEL (Hierarchy Level 2)
                "<text x='0' y='16' font-family='Segoe UI, Semibold' font-size='14' font-weight='700' fill='#34495E'>" & UPPER(CurrentRegion) & "</text>" &
                
                -- BAR
                "<rect x='0' y='28' width='" & BarMaxVWidth & "' height='24' fill='#F0EAE0' rx='2' />" & -- Background Track
                "<rect x='0' y='28' width='" & BarWidth & "' height='24' rx='2' fill='" & BarColor & "' />" &
                
                -- PY MARKER
                IF(YearIsFiltered && NOT(ISBLANK(RevenuePY)), 
                    "<rect x='" & (PYMarkerPos - 1) & "' y='24' width='3' height='32' fill='#2C3E50' />", 
                    ""
                ) &
            "</svg>" &
        "</div>" &

        -- COLUMN 2: Performance Metrics (Hierarchy Level 3)
        "<div style='flex: 0 0 150px; font-family: Segoe UI; text-align: right; border-left: 1px solid #EEE; padding-left: 20px;'>" &
            -- Primary Metric
            "<div style='font-size:18px; font-weight:800; color:#2C3E50; line-height:1;'>" & 
                "$" & FORMAT(DIVIDE(Revenue, Dvsr), "#,0.00") & Sfx & 
            "</div>" &
            
            -- Secondary Details (only shown when filtered)
            IF(YearIsFiltered && NOT(ISBLANK(RevenuePY)), 
                "<div style='margin-top:6px;'>" &
                    "<span style='font-size:11px; font-weight:700; color:#95A5A6; vertical-align:middle;'>PY: $" & FORMAT(DIVIDE(RevenuePY, Dvsr), "#,0.00") & Sfx & "</span>" &
                    "<div style='font-size:13px; font-weight:800; color:" & VarColor & "; margin-top:2px;'>" & 
                        IF(VarPct >= 0, "↑ ", "↓ ") & FORMAT(VarPct, "0.0%") & 
                    "</div>" &
                "</div>",
                ""
            ) &
        "</div>" &
    "</div>"

-- X-AXIS (Hierarchy Level 4)
VAR AX_Val = ABS(GlobalMax)
VAR AX_Sfx = IF(AX_Val >= 1e9, "B", IF(AX_Val >= 1e6, "M", IF(AX_Val >= 1e3, "K", "")))
VAR AX_Dvsr = IF(AX_Val >= 1e9, 1e9, IF(AX_Val >= 1e6, 1e6, IF(AX_Val >= 1e3, 1e3, 1)))

VAR XAxisHTML = 
    "<div style='display: flex; background-color:#FDFBF7; padding: 8px 15px 20px 15px;'>" &
        "<div style='flex: 1;'>" &
            "<svg viewBox='0 0 " & V_Width & " 20' width='100%' preserveAspectRatio='xMinYMid meet'>" &
                "<text x='0' y='15' font-size='10' font-weight='700' fill='#BDC3C7' font-family='Segoe UI'>$0</text>" &
                "<text x='" & BarMaxVWidth & "' y='15' font-size='10' font-weight='700' fill='#BDC3C7' font-family='Segoe UI' text-anchor='end'>MAX SCALE: $" & 
                    FORMAT(DIVIDE(GlobalMax, AX_Dvsr), "#,0.00") & AX_Sfx & "</text>" &
            "</svg>" &
        "</div>" &
        "<div style='flex: 0 0 150px; padding-left: 20px;'></div>" & 
    "</div>"

RETURN 
IF(ISBLANK(Revenue), BLANK(), 
    VAR FirstRegion = MINX(AllRegionsInView, 'Sales'[Region])
    VAR LastRegion = MAXX(AllRegionsInView, 'Sales'[Region])
    RETURN
    SWITCH(TRUE(),
        CurrentRegion = FirstRegion && CurrentRegion = LastRegion, HeaderHTML & ResponsiveRow & XAxisHTML,
        CurrentRegion = FirstRegion, HeaderHTML & ResponsiveRow,
        CurrentRegion = LastRegion, ResponsiveRow & XAxisHTML,
        ResponsiveRow
    )
